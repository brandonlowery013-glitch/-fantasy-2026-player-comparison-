      - name: Run cross-board QA
        run: |
          python - <<'PY'
          import json
          from pathlib import Path
          from collections import Counter,defaultdict

          parts=[]
          for i in range(14):
            x=json.loads(Path(f'players{i}.json').read_text())
            if not isinstance(x,list): raise SystemExit(f'players{i}.json is not a list')
            parts.extend(x)
          if len(parts)!=166: raise SystemExit(f'Expected 166 active players, found {len(parts)}')
          names=[p.get('n') for p in parts]
          dup=[n for n,c in Counter(names).items() if c>1]
          if dup: raise SystemExit(f'Duplicate player names: {dup}')

          ov=json.loads(Path('injuryOverrides2026.json').read_text())
          missing_override=[n for n in ov.get('players',{}) if n not in set(names)]
          if missing_override: raise SystemExit(f'Injury override names missing from 166-player pool: {missing_override}')

          P=[dict(p) for p in parts]
          by={p['n']:p for p in P}
          for n,x in ov.get('players',{}).items(): by[n].update(x)
          for p in P:
            req=['n','p','o','tr','pd','ce','r','e','a','rl','su']
            miss=[k for k in req if p.get(k) is None]
            if miss: raise SystemExit(f"{p['n']} missing required fields {miss}")
            p['s']=round(p['pd']*.35+p['ce']*.20+p['r']*.15+p['e']*.10+p['a']*.10+p['rl']*.05+p['su']*.05,3)

          def apply(field,moves):
            names=list((moves or {}).keys())
            if not names:return
            ordered=sorted(P,key=lambda p:p.get(field,9999))
            picked={p['n']:p for p in ordered if p['n'] in names}
            moved=set(names); slots=[None]*len(ordered)
            for n in names:
              if n in picked:
                t=max(1,min(len(ordered),round(moves[n])))
                if slots[t-1] is not None: raise SystemExit(f'Rank collision at {field} {t}')
                slots[t-1]=picked[n]
            rest=[p for p in ordered if p['n'] not in moved]; j=0
            for i in range(len(slots)):
              if slots[i] is None: slots[i]=rest[j];j+=1
            for i,p in enumerate(slots,1):p[field]=i

          rm=ov.get('rankMoves',{})
          apply('o',rm.get('overall',{})); apply('tr',rm.get('trueValue',{}))
          if sorted(p['o'] for p in P)!=list(range(1,167)): raise SystemExit('Overall ranks are not unique 1-166')
          if sorted(p['tr'] for p in P)!=list(range(1,167)): raise SystemExit('Player Quality ranks are not unique 1-166')

          pos_counts=defaultdict(int)
          for p in P: pos_counts[p['p']]+=1
          if set(pos_counts)!={'QB','RB','WR','TE'}: raise SystemExit(f'Unexpected positions: {dict(pos_counts)}')

          html=Path('index-backup.html').read_text()
          if 'Array.from({length:14}' not in html: raise SystemExit('Runtime loader is not configured for 14 shards')
          if 'Player Quality Rank' not in html: raise SystemExit('Player Quality Rank label missing')
          if 'Expected Fantasy Production' not in html: raise SystemExit('Projection comparison UI missing')
          if 'runtime-rank-movement-context-2026.js' not in html: raise SystemExit('Rank movement runtime is not mounted')

          movement=json.loads(Path('rankMovement2026.json').read_text())
          if movement.get('active_players')!=166: raise SystemExit('Rank movement context is not built against all 166 players')

          qa={
            'updated':'2026-08-31','model':'single 166-player active model','passed':True,
            'active_players':len(P),'unique_players':len(set(names)),'player_shards':14,
            'positions':dict(pos_counts),'missing_required':[],'duplicate_names':[],'missing_override_players':[],
            'overall_rank_range':[min(p['o'] for p in P),max(p['o'] for p in P)],
            'player_quality_rank_range':[min(p['tr'] for p in P),max(p['tr'] for p in P)],
            'runtime_checks':{'loader_14_shards':True,'player_quality_label':True,'projection_comparison_present':True,'rank_movement_context_present':True,'season_phase_contract_present':True,'cache_bust':stamp if False else '20260831-166-phase'},
            'admissions':{n:{k:by[n].get(k) for k in ['p','o','tr','a','st','px','s7','mp']} for n in ['Kaleb Johnson','Corey Kiner','Tank Dell','Jonnu Smith']}
          }
          Path('final-cross-board-qa-2026-08-24.json').write_text(json.dumps(qa,indent=2,ensure_ascii=False)+'\n')

          a=Path('comparison-sync-162-audit.json')
          old=json.loads(a.read_text()); old['updated']='2026-08-31'; old['active_players']=166; old['unique_normalized']=166
          old['missing_required']=[]; old['alias_duplicates']=[]
          old['runtime']={'cross_board_qa_passed':True,'player_loader_shards':14,'active_players':166,'rank_movement_context':True,'season_phase_contract':True,'cache_bust':'20260831-166-phase'}
          a.write_text(json.dumps(old,indent=2,ensure_ascii=False)+'\n')
          PY
      - name: Validate JavaScript syntax
        run: |
